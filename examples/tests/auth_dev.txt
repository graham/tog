# Dev Authentication Tests
# Tests the /dev/create_and_assume endpoint for creating test users
# Session persistence is enabled by default with -f

# Create a new user and assume their identity
/dev/create_and_assume?email=devtest@example.com
?assert status 200
?assert json .email "devtest@example.com"
?assert json .message "session created"

# Verify the session cookie was set
?session

# Now we should be authenticated - verify with whoami
/auth/whoami
?assert status 200
?assert json .authenticated true
?assert json .email "devtest@example.com"
?assert json .is_admin false

# Create another user with different email
/dev/create_and_assume?email=another@example.com
?assert status 200
?assert json .email "another@example.com"

# Session should now be for the new user
/auth/whoami
?assert status 200
?assert json .email "another@example.com"
